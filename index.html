まずはJavaScriptの基本から。

学習の入口0件表示

console.log()

コンソールに値を表示

console.log('Hello');

const

あとから再代入しない変数を作る

const name = 'Tomo';

let

あとから値を変更できる変数を作る

let count = 0;
count = 1;

if / else

条件によって処理を分ける

if (score >= 60) {
  console.log('合格');
} else {
  console.log('不合格');
}

for

回数を決めて繰り返す

for (let i = 0; i < 5; i++) {
  console.log(i);
}

for...of

配列などの値を1つずつ取り出す

for (const item of items) {
  console.log(item);
}

function

処理をまとめて関数を作る

function add(a, b) {
  return a + b;
}

アロー関数

functionを短く書く方法

  return a + b;
};

return

function double(n) {
  return n * 2;
}

map()

filter()

条件に合う要素だけを取り出す

const nums = [1, 2, 3, 4];
const even = nums.filter(n => n % 2 === 0);

find()

条件に合う最初の要素を探す

const user = users.find(u => u.id === 1);

forEach()

配列の各要素に処理を行う

  console.log(item);
});

分割代入

スプレッド構文 ...

  console.error(error);

import / export

別ファイルの機能を読み込む・公開する

===

値と型の両方が等しいか比較する

if (age === 18) {}

!==

値または型が等しくないか比較する

if (age !== 18) {}

&&

両方の条件がtrueならtrue

if (age >= 18 && hasId) {}

||

どちらか一方がtrueならtrue

const name = input || '名無し';

!

true / falseを反転する

if (!isLogin) {}
